4.3 Socket.IO 시그널링 클라이언트

class SignallingClient( private val serverUrl: String = "http://your-server:3004" ) { private lateinit var socket: Socket private val gson = Gson() var onOfferReceived: ((SessionDescription) -> Unit)? = null var onAnswerReceived: ((SessionDescription) -> Unit)? = null var onIceCandidateReceived: ((IceCandidate) -> Unit)? = null fun connect(authToken: String) { val options = IO.Options().apply { auth = mapOf("token" to authToken) transports = arrayOf("websocket") } socket = IO.socket(serverUrl, options) setupEventListeners() socket.connect() } private fun setupEventListeners() { socket.on("offer") { args -> val data = args[0] as JSONObject val offer = SessionDescription( SessionDescription.Type.OFFER, data.getJSONObject("offer").getString("sdp") ) onOfferReceived?.invoke(offer) } socket.on("answer") { args -> val data = args[0] as JSONObject val answer = SessionDescription( SessionDescription.Type.ANSWER, data.getJSONObject("answer").getString("sdp") ) onAnswerReceived?.invoke(answer) } socket.on("ice-candidate") { args -> val data = args[0] as JSONObject val candidateData = data.getJSONObject("candidate") val candidate = IceCandidate( candidateData.getString("sdpMid"), candidateData.getInt("sdpMLineIndex"), candidateData.getString("candidate") ) onIceCandidateReceived?.invoke(candidate) } } fun joinRoom(roomId: String) { socket.emit("join-room", roomId) } fun sendOffer(offer: SessionDescription, targetUserId: String) { val json = JSONObject().apply { put("targetUserId", targetUserId) put("offer", JSONObject().apply { put("type", offer.type.toString()) put("sdp", offer.description) }) } socket.emit("offer", json) } }

class CallService : Service() { private lateinit var notificationManager: CallNotificationManager private var callTimer: Timer? = null private var callDuration = 0 override fun onCreate() { super.onCreate() notificationManager = CallNotificationManager(this) } override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int { when (intent?.action) { ACTION_START_CALL -> startCall() ACTION_END_CALL -> endCall() ACTION_MUTE -> toggleMute() } return START_STICKY } private fun startCall() { // 포그라운드 서비스 시작 val notification = notificationManager.createCallNotification( "통화 중", "00:00" ) startForeground(NOTIFICATION_ID, notification) // 통화 타이머 시작 startCallTimer() } private fun startCallTimer() { callTimer = Timer() callTimer?.scheduleAtFixedRate(object : TimerTask() { override fun run() { callDuration++ updateNotification() } }, 0, 1000) } private fun updateNotification() { val minutes = callDuration / 60 val seconds = callDuration % 60 val timeString = String.format("%02d:%02d", minutes, seconds) val notification = notificationManager.createCallNotification( "통화 중", timeString ) notificationManager.notify(NOTIFICATION_ID, notification) } }
